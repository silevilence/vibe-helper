const TEST_LABELS: Record<string, string> = {
  tdd: 'TDD（测试驱动开发）',

};
 
/** 权限标签 */
const PERM_LABELS: Record<string, string> = {
  deny: '禁止',
  allow: '允许',
  'manual-only': '仅限手动',
